Docker启动PostgreSQL服务：

Image:
 docker pull postgres:16
Container:
 docker run -d --name harness-postgres 
 -e POSTGRES_USER=fluctuate_user
 -e POSTGRES_PASSWORD=123456 
 -e POSTGRES_DB=fluctuate_db 
 -p 5432:5432 
 -v harness_pgdata:/var/lib/postgresql/data 
 postgres:16
